// HitZone.cs — 通用受击组件，角色和怪物共用
[RequireComponent(typeof(Collider))]
class HitZone : MonoBehaviour {
    IDamageable _owner;
    HashSet<int> _hitInstanceIds; // 用 instanceID 去重

    public void Init(IDamageable owner) { _owner = owner; }

    void OnTriggerStay(Collider other) {
        var hitbox = other.GetComponent<IAttackHitbox>();
        if (hitbox == null || !hitbox.IsActive) return;
        if (!_hitInstanceIds.Add(hitbox.GetInstanceID())) return; // 已有
        _owner.TakeDamage(hitbox.CurrentData.DamageData,
            (transform.position - hitbox.GetBounds().center).normalized);
    }
}
AttackResolver — 静态工具类，角色和怪物攻击都用它
// 攻击检测统一入口，NonAlloc 无 GC
public static class AttackResolver
{
    static readonly Collider[] _buffer = new Collider[16]; // 复用缓冲区

    /// 在前方扇形区域内查找最近的可伤害目标
    public static IDamageable FindNearestTarget(
        Vector3 origin, Vector3 forward, float range, float angle, LayerMask mask)
    {
        int count = Physics.OverlapSphereNonAlloc(origin, range, _buffer, mask);
        IDamageable best = null;
        float bestDist = float.MaxValue;

        for (int i = 0; i < count; i++)
        {
            var col = _buffer[i];
            Vector3 dir = col.bounds.center - origin;
            float dist = dir.magnitude;
            if (dist > range) continue;

            float angleToTarget = Vector3.Angle(forward, dir);
            if (angleToTarget > angle * 0.5f) continue;

            var target = col.GetComponentInParent<IDamageable>();
            if (target == null || !target.IsAlive) continue;

            if (dist < bestDist) { best = target; bestDist = dist; }
        }
        return best;
    }
}

优化 3：AttackEffect 管线

// 替代 DamageData 的单维模型，支持一次攻击挂载多种效果
[Serializable]
public class AttackEffectConfig
{
    public DamageData Damage;           // 基础伤害
    public float KnockbackForce;       // 击退
    public float StunDuration;         // 眩晕
    public StatusEffectType Status;    // 状态效果 (Poison/Bleed/Slow/None)
    public float StatusDuration;
    public float StatusValue;          // 效果数值 (如减速30%)
}

// MonsterConfig 中
public AttackEffectConfig[] AttackEffects; // 每个攻击对应一组效果

// 使用时
foreach (var effect in config.AttackEffects)
{
    target.ApplyDamage(effect.Damage);
    target.ApplyKnockback(effect.KnockbackForce);
    target.ApplyStatus(effect.Status, effect.StatusDuration, effect.StatusValue);
}
