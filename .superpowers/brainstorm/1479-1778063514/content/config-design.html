MonsterAI 改造要点

Attack 行为：配置驱动
// MonsterAI.TransitionTo(Attack) 中
if (_config.AttackAnimCount > 1)
{
    // 按 AttackWeights 权重随机选择攻击
    int index = WeightedRandom(_config.AttackWeights);
    _currentAttackIndex = index;
    _animator.SetInteger("AttackIndex", index);
    var dmg = _config.AttackDamageList[index];
    OnAttackFrame?.Invoke(dmg);
}
else
{
    _animator.SetTrigger(HASH_Attack);
    OnAttackFrame?.Invoke(_config.AttackDamage);
}
_animator.SetTrigger(HASH_Attack);
Defend 行为：配置驱动
// MonsterAI.EvaluateTransitions() 中
if (_config.EnableDefend &&
    _stats.HP / _stats.MaxHP < _config.DefendHPThreshold
    && _defendCooldown <= 0)
{
    TransitionTo(MonsterAIState.Defend);
}

// ExecuteState(Defend) 中
if (_stateTimer <= 0)
{
    if (_defendBlockCount >= _config.DefendBlockCountToCounter)
    {
        // 触发反击
        _counterAttackActive = true;
        TransitionTo(MonsterAIState.Attack);
    }
    else
        TransitionTo(MonsterAIState.Chase);
}

MonsterAI 新增状态

public enum MonsterAIState
{
    Idle    = 0,
    Patrol  = 1,
    Chase   = 2,
    Attack  = 3,
    Hit     = 4,
    Death   = 5,
    Defend  = 6,   // 新增：防御状态
    Taunt   = 7,   // 新增：嘲讽状态 (Slime)
    Alert   = 8,   // 新增：警戒状态
}
