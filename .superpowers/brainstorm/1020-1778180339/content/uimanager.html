UIManager — Panel Management

单例 / 5 层 Canvas 管理 / Stack + Overlay / 异步动画协调

Core API

// Registration (called once on init)
public void Register(UIPanel panel);

// Stack panels (Main / Popup layers)
public async UniTask PushAsync(string panelId);  // show + push to stack
public async UniTask PopAsync();               // hide top + pop from stack
public void PopTo(string panelId);         // pop until panelId is top

// Overlay panels (Top / Guide layers)
public async UniTask OpenAsync(string panelId);
public async UniTask CloseAsync(string panelId);

// Always panels (Base layer — HUD etc)
public async UniTask ShowAlwaysAsync(string panelId);
public async UniTask HideAlwaysAsync(string panelId);
Stack Behavior
Push("BagPanel"):
→ BagPanel.OnPreShow()
→ BagPanel.PlayShowAnimation()
→ BagPanel.OnShow()
→ stack: [BagPanel]

Push("ItemDetail"):
→ BagPanel.OnHide() (pause events)
→ ItemDetail.OnPreShow()
→ ItemDetail.PlayShowAnimation()
→ ItemDetail.OnShow()
→ stack: [BagPanel, ItemDetail]

Pop():
→ ItemDetail.OnHide() (stop events)
→ ItemDetail.PlayHideAnimation() (visual out)
→ ItemDetail visibility OFF
→ BagPanel.OnShow() (resume events)
→ stack: [BagPanel]
Overlay Behavior
Open("ReconnectBtn"):
→ OnPreShow → ShowAnim → OnShow
→ stack: unaffected
→ overlay: [ReconnectBtn]

Open("Loading"):
→ OnPreShow → ShowAnim → OnShow
→ stack: unaffected
→ overlay: [ReconnectBtn, Loading]

Close("ReconnectBtn"):
→ OnHide → HideAnim → vis OFF
→ overlay: [Loading]

Note:
Overlay panels are managed independently
Multiple overlays can display at once
Order is by Canvas sort (Top=4000, Guide=5000)

Internal Structure

public class UIManager : MonoBehaviour
{
  // Singleton
  public static UIManager Instance { get; private set; }

  // 5 Canvas roots (created on init)
  private Canvas[] _layers = new Canvas[5];

  // Panel registry: all registered panels by ID
  private Dictionary<string, UIPanel> _registry;

  // Stack for Main/Popup layers
  private Stack<UIPanel> _stack;

  // Active overlays
  private HashSet<UIPanel> _activeOverlays;
}
