UIAnimation — DOTween Animation System

组件化设计 / Sequence Builder / 预设动画 / 状态机扩展

UIAnimation Component (attached to UIPanel GameObject)

public class UIAnimation : MonoBehaviour
{
  // Preset animations (configured in Inspector or code)
  public UIAnimPreset ShowPreset;  // fade-in + scale:0.9→1.0
  public UIAnimPreset HidePreset;  // fade-out + scale:1.0→0.95

  // Build custom sequences
  public SequenceBuilder Build() → new SequenceBuilder(transform);

  // Play presets (called by UIManager)
  public Sequence PlayShow();
  public Sequence PlayHide();

  // Direct tween helpers
  public Tweener FadeIn(float duration = 0.3f);
  public Tweener ScaleIn(float duration = 0.3f);
  public Tweener SlideIn(Direction dir, float duration = 0.3f);

  // Kill all running tweens on this object
  public void KillAll();
}

SequenceBuilder (Fluent API)
var seq = anim.Build()
  .FadeIn(0.2f)      // alpha 0→1
  .Join(() => anim.ScaleIn(0.3f)) // + scale 0.9→1
  .Then(() => icon.Punch())  // then icon bounce
  .Delay(0.1f)      // small pause
  .Then(() => text.CountUp(100)) // numeric count
  .Play();         // → Sequence
