AIStateBase — The Key Abstraction

// Design: Each AI state is a self-contained class with clear lifecycle.
// This makes adding a new state (e.g., BossPhase2Idle) trivial —
// just subclass and override the hooks you need.
//
// To add a new state type in the future:
// 1. Create a subclass (e.g., BossEnrageState)
// 2. Register it in AIStateMachine's transition table
// 3. Done — no need to touch AIBrain or any other state.

public abstract class AIStateBase
{
    // Called once when entering this state.
    // Override to set animator params, start timers, etc.
    public virtual void OnEnter(AIContext ctx) { }

    // Called every frame while this state is active.
    // Return the next state to transition to, or null to stay.
    // Override to define per-state transition logic.
    public virtual AIStateBase EvaluateTransitions(AIContext ctx) { return null; }

    // Called every frame. Override for per-frame behavior (movement, look-at, etc.)
    public virtual void OnUpdate(AIContext ctx) { }

    // Called once when leaving this state. Clean up animator params, timers.
    public virtual void OnExit(AIContext ctx) { }

    // Unique identifier for animator hash lookup.
    public abstract MonsterAIState StateType { get; }
}

AIBrain — The Orchestrator

// AIBrain is the single entry point for monster AI.
// It owns the AIContext, AIStateMachine, and behaviours.
// MonsterEntity calls Update() each frame — everything else is internal.
//
// Design principle: AIBrain knows nothing about specific states or behaviours.
// States register themselves in the state machine. Behaviours are scheduled
// automatically based on active state. Adding a new state or behaviour
// requires zero changes to this class.

public class AIBrain
{
    private readonly AIContext _ctx;           // Shared mutable state (class)
    private readonly AIStateMachine _fsm;
    private readonly BehaviourScheduler _behaviours;

    public AIBrain(MonsterConfig config, Transform self, ...) { ... }

    public void Update(float deltaTime)
    {
        if (_ctx.IsDead) return;

        _ctx.DeltaTime = deltaTime;
        _ctx.Movement.UpdateKnockback(deltaTime);

        TryFindTarget();
        _behaviours.UpdateActive(_ctx);   // Calls Update() on active behaviours
        _fsm.Update(_ctx);                // Evaluate transitions + execute state
    }

    // Called by MonsterEntity.TakeDamage (via damage pipeline)
    public void NotifyHit(DamageResult result, Vector3 hitDirection)
    {
        _ctx.LastHitResult = result;
        _ctx.LastHitDirection = hitDirection;
        _fsm.OnHitReceived(_ctx);  // FSM decides: enter Hit state or ignore
    }
}
