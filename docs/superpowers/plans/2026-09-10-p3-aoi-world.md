# P3 AOI 大世界化 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [x]`) syntax for tracking.

**Goal:** 九宫格 AoiGrid 替换全量可见占位：可见实体按距离带分级更新（近 15Hz/中 8Hz/远 4Hz），视野进出驱动远程实体生成/销毁（客户端缺席超时 + 迟滞防抖），BotRunner 压测 100+ bot，带宽门禁实测达标。

**Architecture:** 服务器广播从 store 级（dirty+beacon 一次组装全体复用）重构为 **per-observer 分级组装**（AoiGrid 按观察者坐标 ±视野半径粗筛候选格 + 精确距离迟滞过滤；AoiBandScheduler 节拍：**移动实体按距离带频率，静止实体统一 4Hz 保活**——保活与客户端缺席超时语义闭合）；EntityStore 位置为唯一事实源，脏标记保留但不再驱动广播节拍。客户端 RemoteEntityRegistry 增加缺席超时删除（AOI 下视野外实体自然停流 → 1.2s 无快照即销毁），协议零改动。

**Tech Stack:** .NET 8 / Unity 2022.3（C#9）/ protobuf-net 3.2.56 / xUnit + NUnit(EditMode) + PlayMode

**Spec:** `docs/superpowers/specs/2026-08-27-network-sync-design.md` §5-P3 + D5（坐标量化条件项）

```csharp
using System.Collections.Generic;

namespace KcpServer
{
    /// <summary>
    /// 网格 AOI（spec §5-P3）——按观察者坐标 ±SightRadius 推导候选格范围（粗筛），
    /// 精确半径过滤 + 迟滞边界。**网格是过滤器而非索引**：每 observer 每 tick 仍遍历
    /// allPlayers 一次（O(N)，数百实体毫秒级——性能目标为"不随世界面积增长"，由粗筛保证）。
    /// 迟滞状态按 observer 维度存 _sightState；实体注销经 RemoveEntity 清除（playerId 单调
    /// 递增不重用，但登录-注销-再登录同 id 只在 RemoveEntity 缺失时产生旧状态残留——见
    /// RemoveEntity_Clears_Observer_Sight_State 测试）。
    /// </summary>
    public sealed class AoiGrid
    {
        public const float CellSize = 40f;
        public const float SightRadius = 80f;
        public const float LeaveRadius = 90f; // 迟滞：进 ≤80m，出须 >90m

        private readonly Dictionary<long, Dictionary<long, bool>> _sightState =
            new Dictionary<long, Dictionary<long, bool>>();

        /// <summary>观察者可见实体列表（实体自身；顺序 = allPlayers 遍历序，调用方不依赖）</summary>
        public List<PlayerState> GetVisible(long observerId, IReadOnlyCollection<PlayerState> allPlayers)
        {
            var result = new List<PlayerState>();
            if (!_sightState.TryGetValue(observerId, out var seen))
            {
                seen = new Dictionary<long, bool>();
                _sightState[observerId] = seen;
            }

            // 观察者当前位置（找不到观察者 → 空视野）
            PlayerState? observer = null;
            foreach (var p in allPlayers)
                if (p.PlayerId == observerId) { observer = p; break; }
            if (observer == null) return result;

            float ox = observer.Position.X, oz = observer.Position.Z;
            // 候选格范围由半径从坐标推导。范围必须 ≥ LeaveRadius(90) 而非 SightRadius(80)：
            // ① 迟滞环（80~90m）实体须持续被评估（坐标 mod 40 ∈ (30,40) 时 ±80 只覆盖
            //    到 80~120−u < 90——出界判定漏评估 → 停流早于 90m + seen 残留 ghost；
            //    复核 R1，此即 D2 语义洞）；
            // ② 若实体在候选范围外离开（>90m）seen 不更新——回落到 80~90m 时凭 stale-true
            //    错误地"仍可见"（重进须 ≤80）。
            // 非"所在格 ±1"（后者短边盲区，终审 C2）。±90 覆盖 ≤5×5=25 格候选。
            int colMin = CellOf(ox - LeaveRadius), colMax = CellOf(ox + LeaveRadius);
            int rowMin = CellOf(oz - LeaveRadius), rowMax = CellOf(oz + LeaveRadius);

            foreach (var p in allPlayers)
            {
                if (p.PlayerId == observerId) continue;
                int col = CellOf(p.Position.X), row = CellOf(p.Position.Z);
                if (col < colMin || col > colMax || row < rowMin || row > rowMax) continue;

                float dx = p.Position.X - ox, dz = p.Position.Z - oz;
                float distSqr = dx * dx + dz * dz;
                bool wasVisible = seen.TryGetValue(p.PlayerId, out var v) && v;
                bool visibleNow = distSqr <= SightRadius * SightRadius
                    || (wasVisible && distSqr <= LeaveRadius * LeaveRadius);
                seen[p.PlayerId] = visibleNow;
                if (visibleNow) result.Add(p);
            }

            return result;
        }

        /// <summary>实体注销——从所有观察者视野状态清除（防同 id 重登残留迟滞豁免）</summary>
        public void RemoveEntity(long entityId)
        {
            foreach (var seen in _sightState.Values)
                seen.Remove(entityId);
        }

        /// <summary>移除观察者全部视野状态（会话断开时由 Server3C 调用）</summary>
        public void RemoveObserver(long observerId) => _sightState.Remove(observerId);

        private static int CellOf(float coord)
        {
            // 负坐标向下取整：-45 → cell -2（-40..0 是 -1）
            return coord >= 0f ? (int)(coord / CellSize) : (int)(coord / CellSize) - 1;
        }
    }
}
```

注意：`CellOf` 负坐标处理——`(int)(-45f / 40f)` = `(int)(-1.125)` = -1（C# 向零截断），需再 -1 → -2 ✓（-80..-40 为 -2）。验证：coord=-39 → -0.975 → (int)=-0…`(int)(-0.975)` = 0，-1 → -1 ✓（-40..0 是 -1 格）。

```csharp
using Xunit;

namespace KcpServer.Tests;

public class AoiBandSchedulerTests
{
    // 统一签名：ShouldSend(observerId, entityId, distance, tick, x, z, rotY)
    // 语义（终审 I5 修正）：移动实体（坐标自上次发送变化）按带频；
    // 静止实体（坐标未变）统一 4 tick 保活——近带静止 15Hz 全字段是带宽回归。
    private static bool Send(AoiBandScheduler s, long o, long e, float dist, uint tick,
        float x = 0f, float z = 0f, float rot = 0f)
        => s.ShouldSend(o, e, dist, tick, x, z, rot);

    [Fact]
    public void Near_Band_Moving_Entity_Updates_Every_Tick()
    {
        var scheduler = new AoiBandScheduler();
        // 移动中（x 每 tick +1）：近带 29m → tick 1,2,3... 全发
        Assert.True(Send(scheduler, 1, 2, 29f, 1, x: 10f));
        Assert.True(Send(scheduler, 1, 2, 29f, 2, x: 11f));
        Assert.True(Send(scheduler, 1, 2, 29f, 3, x: 12f));
    }

    [Fact]
    public void Mid_Band_Moving_Entity_Updates_Every_Second_Tick()
    {
        var scheduler = new AoiBandScheduler();
        Assert.True(Send(scheduler, 1, 2, 45f, 1, x: 10f));
        Assert.False(Send(scheduler, 1, 2, 45f, 2, x: 11f)); // 间隔未到
        Assert.True(Send(scheduler, 1, 2, 45f, 3, x: 12f));
        Assert.False(Send(scheduler, 1, 2, 45f, 4, x: 13f));
    }

    [Fact]
    public void Far_Band_Moving_Entity_Updates_Every_Fourth_Tick()
    {
        var scheduler = new AoiBandScheduler();
        Assert.True(Send(scheduler, 1, 2, 70f, 1, x: 10f));
        Assert.False(Send(scheduler, 1, 2, 70f, 2, x: 11f));
        Assert.False(Send(scheduler, 1, 2, 70f, 3, x: 12f));
        Assert.False(Send(scheduler, 1, 2, 70f, 4, x: 13f));
        Assert.True(Send(scheduler, 1, 2, 70f, 5, x: 14f));
    }

    [Fact]
    public void Static_Entity_Keepalive_Every_Four_Ticks_In_Any_Band()
    {
        // 静止（坐标恒同）近带实体：首帧后仅每 4 tick 保活——不占每 tick 带宽
        var scheduler = new AoiBandScheduler();
        Assert.True(Send(scheduler, 1, 2, 10f, 1));   // 首帧
        Assert.False(Send(scheduler, 1, 2, 10f, 2));  // 静止：距上次 < 4 tick
        Assert.False(Send(scheduler, 1, 2, 10f, 3));
```

```csharp
using System.Collections.Generic;

namespace KcpServer
{
    /// <summary>
    /// 距离带分级节拍（spec §5-P3：近 15Hz/中 8Hz/远 4Hz）——
    /// 移动实体（坐标/旋转自上次发送变化）按带频广播；静止实体统一 4 tick 保活
    /// （近带静止 15Hz 全字段是带宽回归——终审 I5 修正；4 tick ≈ 267ms ≪ 客户端 1.2s 缺席超时）。
    /// 每个 (observer, entity) 记录最近发送 tick、距离带与上次坐标。
    /// 带变化/重见即重置（立即发一帧）。主循环独占线程，无锁。
    /// tick 差值用相减（uint wrap 安全：15Hz 下 ~9.1 年才 wrap）。
    /// </summary>
    public sealed class AoiBandScheduler
    {
        public const float MidBandStart = 30f;
        public const float FarBandStart = 60f;
        public const uint StaticKeepaliveInterval = 4; // 静止实体统一 4 tick 保活

        private struct Slot
        {
            public uint LastSentTick;
            public int Band; // 0=近 1=中 2=远
            public float X;
            public float Z;
            public float RotY;
            public bool HasSent;
        }

        private readonly Dictionary<long, Dictionary<long, Slot>> _state =
            new Dictionary<long, Dictionary<long, Slot>>();

        /// <summary>distance 米、tick 当前服务器 tick、x/z/rotY 实体当前坐标。
        /// 返回 true = 本 tick 应给该观察者发该实体</summary>
        public bool ShouldSend(long observerId, long entityId, float distance, uint tick,
            float x, float z, float rotY)
        {
            int band = BandOf(distance);
            uint interval = IntervalOf(band);

            if (!_state.TryGetValue(observerId, out var entities))
            {
                entities = new Dictionary<long, Slot>();
                _state[observerId] = entities;
            }

            if (!entities.TryGetValue(entityId, out var slot) || !slot.HasSent)
            {
                entities[entityId] = new Slot { LastSentTick = tick, Band = band, X = x, Z = z, RotY = rotY, HasSent = true };
                return true; // 首帧（或带变化/重见后）立即发
            }

            if (slot.Band != band)
            {
                entities[entityId] = new Slot { LastSentTick = tick, Band = band, X = x, Z = z, RotY = rotY, HasSent = true };
                return true; // 带变化：重置节拍
            }

            bool moved = x != slot.X || z != slot.Z || rotY != slot.RotY;
            uint elapsed = tick - slot.LastSentTick;
            bool due = moved ? elapsed >= interval : elapsed >= StaticKeepaliveInterval;

            if (due)
            {
                entities[entityId] = new Slot { LastSentTick = tick, Band = band, X = x, Z = z, RotY = rotY, HasSent = true };
                return true;
            }
            return false;
        }

        /// <summary>实体离开视野时调用——清除节拍，重新入场从零开始</summary>
        public void MarkOutOfSight(long observerId, long entityId)
        {
            if (_state.TryGetValue(observerId, out var entities))
                entities.Remove(entityId);
        }

        /// <summary>实体注销——从所有观察者节拍状态清除</summary>
        public void RemoveEntity(long entityId)
        {
            foreach (var entities in _state.Values)
                entities.Remove(entityId);
        }

        /// <summary>移除观察者全部节拍状态（会话断开）</summary>
        public void RemoveObserver(long observerId) => _state.Remove(observerId);

        /// <summary>按距离返回带编号：0 近(&lt;30) / 1 中(30~60) / 2 远(60~80)</summary>
        public static int BandOf(float distance)
        {
            if (distance < MidBandStart) return 0;
            if (distance < FarBandStart) return 1;
            return 2;
        }

        /// <summary>带间隔（tick 数）：近 1 / 中 2 / 远 4</summary>
        public static uint IntervalOf(int band) => band switch
        {
            0 => 1,
            1 => 2,
            _ => 4
        };
    }
}
```

```csharp
using System.Collections.Generic;
using Cool.Protocol;

namespace KcpServer
{
    /// <summary>
    /// per-observer 分级广播组装（spec §5-P3）——取代 SnapshotAssembler 的
    /// store 级 dirty/beacon 广播：每个 observer 每 tick 经 AoiGrid 取可见集、
    /// AoiBandScheduler 按距离带节拍 + 移动/静止判定选出应发实体，组装当前全字段快照。
    /// 发送由 Server3C 注入（OnEntityBroadcast）；本类纯组装可单测。
    /// 主循环独占线程，无锁。实体注销经 RemoveEntity 清理 AOI 状态。
    /// </summary>
    public sealed class AoiBroadcaster
    {
        private readonly AoiGrid _grid = new AoiGrid();
        private readonly AoiBandScheduler _scheduler = new AoiBandScheduler();

        /// <summary>每发一个实体快照回调（Server3C 注入：写入该 observer 的 WorldSnapshot.Entities）</summary>
        public System.Action<long, EntitySnapshot>? OnEntityBroadcast { get; set; }

        /// <summary>
        /// 为观察者组装本 tick 应广播的实体快照列表（AoiGrid 可见 + 带节拍/移动判定）。
        /// 每 tick 每 observer 调用一次。语义：移动实体按带频、静止实体 4 tick 保活
        /// （决策 D4——客户端缺席超时 1.2s 靠 4Hz 保活闭合）。
        /// </summary>
        public List<EntitySnapshot> BuildForObserver(PlayerState observer, uint tick, EntityStore store)
        {
            var result = new List<EntitySnapshot>();
            float ox = observer.Position.X, oz = observer.Position.Z;

            var visible = _grid.GetVisible(observer.PlayerId, store.All);
            foreach (var entity in visible)
            {
                float dx = entity.Position.X - ox, dz = entity.Position.Z - oz;
                float distance = (float)System.Math.Sqrt(dx * dx + dz * dz);

                if (_scheduler.ShouldSend(observer.PlayerId, entity.PlayerId, distance, tick,
                    entity.Position.X, entity.Position.Z, entity.Rotation))
                {
                    var snap = new EntitySnapshot
                    {
                        PlayerId = entity.PlayerId,
                        FieldMask = (ushort)(EntityFieldMask.Position | EntityFieldMask.Rotation | EntityFieldMask.Speed),
                        X = entity.Position.X,
                        Y = entity.Position.Y,
                        Z = entity.Position.Z,
                        Rotation = entity.Rotation,
                        Speed = entity.Speed,
                        ActionState = 0,
                        Tick = tick
                    };
                    result.Add(snap);
                    OnEntityBroadcast?.Invoke(entity.PlayerId, snap);
                }
            }
            return result;
        }

        /// <summary>实体注销——清理其全部 AOI 状态（视野迟滞 + 节拍槽）</summary>
        public void RemoveEntity(long entityId)
        {
            _grid.RemoveEntity(entityId);
            _scheduler.RemoveEntity(entityId);
        }

        /// <summary>观察者会话断开——清理其 AOI 状态</summary>
        public void RemoveObserver(long observerId)
        {
            _grid.RemoveObserver(observerId);
            _scheduler.RemoveObserver(observerId);
        }
    }
}
```

**注意**：实体离开视野（迟滞出界）后其节拍槽保留——重见时 `tick - LastSentTick` 必 ≥ 间隔 → 立即发（AoiBandScheduler 测试 Out_Of_Sight_Clears_Schedule 验证 MarkOutOfSight 语义仍工作；广播路径不强制调用，靠 elapsed 兜底等价）。槽位泄漏由 RemoveEntity（注销路径）与 RemoveObserver（断开路径）收敛——每槽 ~20B，数百实体规模可忽略。

```csharp
using System;
using System.Collections.Generic;

namespace KcpServer
{
    /// <summary>
    /// 压测 bot 群（spec §5-P3：100~200 移动 bot）——注册为 EntityStore 假玩家
    /// （无 session，只被观察者看），每 tick 直线游走 + 边界反弹。确定性随机
    /// （固定种子）保证测试可复现。主循环线程独占调用。
    /// </summary>
    public sealed class BotRunner
    {
        public const float WorldHalfExtent = 500f;  // ±500m 世界
        public const float BotSpeedPerTick = 0.53f; // ~8m/s @ 66ms tick

        private readonly EntityStore _store;
        private readonly Func<long> _nextPlayerId;
        private readonly Random _random;
        private readonly List<Bot> _bots = new List<Bot>();
        private long _tickCounter;

        private sealed class Bot
        {
            public long PlayerId;
            public float DirX;
            public float DirZ;
        }

        public BotRunner(EntityStore store, Func<long> nextPlayerId, int seed = 12345)
        {
            _store = store;
            _nextPlayerId = nextPlayerId;
            _random = new Random(seed);
        }

        /// <summary>生成 count 个 bot（随机位置 + 随机方向）</summary>
        public void Spawn(int count)
        {
            for (int i = 0; i < count; i++)
            {
                long id = _nextPlayerId();
                float x = (float)(_random.NextDouble() * 2 - 1) * WorldHalfExtent * 0.8f;
                float z = (float)(_random.NextDouble() * 2 - 1) * WorldHalfExtent * 0.8f;
                var player = _store.AddPlayer(id, $"Bot_{id}");
                player.UpdatePosition(new Vector3(x, 0, z), 0f, 0f, 1L, 0U);
                _store.MarkDirty(player); // 注册即标脏——首个快照携带（保 P2 语义）
                var bot = new Bot { PlayerId = id };
                RandomDir(bot);
                _bots.Add(bot);
            }
        }

        public void DespawnAll()
        {
            foreach (var bot in _bots)
                _store.RemovePlayer(bot.PlayerId);
            _bots.Clear();
        }

        /// <summary>推进一 tick（服务器主循环每 tick 调用）</summary>
        public void Tick(uint tick)
        {
            _tickCounter++;
            foreach (var bot in _bots)
            {
                var player = _store.GetPlayer(bot.PlayerId);
                if (player == null) continue;

                float x = player.Position.X + bot.DirX * BotSpeedPerTick;
                float z = player.Position.Z + bot.DirZ * BotSpeedPerTick;

                // 边界反弹
                if (x > WorldHalfExtent || x < -WorldHalfExtent) { bot.DirX = -bot.DirX; x = Clamp(x); }
                if (z > WorldHalfExtent || z < -WorldHalfExtent) { bot.DirZ = -bot.DirZ; z = Clamp(z); }

                player.UpdatePosition(new Vector3(x, 0, z), YawOf(bot.DirX, bot.DirZ), 8f, tick * 10_000L + 1L, (uint)_tickCounter);
            }
        }

        private static float Clamp(float v) => Math.Clamp(v, -WorldHalfExtent, WorldHalfExtent);

        private static float YawOf(float dx, float dz) => (float)(System.Math.Atan2(dx, dz) * 180.0 / System.Math.PI);

        /// <summary>随机四向移动（单轴 ±1——保持 ~8m/s 标称速度；双向斜走是 √2× 超速）</summary>
        private void RandomDir(Bot bot)
        {
            if (_random.NextDouble() < 0.5)
            {
                bot.DirX = _random.NextDouble() < 0.5 ? 1f : -1f;
                bot.DirZ = 0f;
            }
            else
            {
                bot.DirX = 0f;
                bot.DirZ = _random.NextDouble() < 0.5 ? 1f : -1f;
            }
        }
    }
}
```

```csharp
namespace KcpNet.PlayModeTests

    /// P3 核心验收：bot 走近（客户端生成视图）→ bot 走远（缺席超时销毁，无抖动残留）；

    /// </summary>

    {
        private sealed class Ctx
        {
            public KcpNetClientAdapter Adapter = new KcpNetClientAdapter();
            public ClientNetworkManager Manager = null!;
            public RemoteEntityRegistry Registry = null!;
        }

        [UnityTest]
        public IEnumerator Bots_Enter_And_Leave_Sight_Over_Time()
        {
            var runTask = RunAsync();
            float elapsed = 0f;
            while (!runTask.IsCompleted && elapsed < 120f)
            {
                elapsed += Time.deltaTime;
                yield return null;
            }
            Assert.IsTrue(runTask.IsCompleted, "aoi test timed out");
            var (ok, reason) = runTask.Result;
            Assert.IsTrue(ok, reason);
        }

        private static async System.Threading.Tasks.Task<(bool, string)> RunAsync()
        {
            var root = new GameObject("AoiRoot");
            var registry = new RemoteEntityRegistry(root.transform);
            var ctx = new Ctx();
            ctx.Registry = registry;
            ctx.Manager = new ClientNetworkManager(ctx.Adapter, registry, "Aoi_Observer");
            try
            {
                ctx.Manager.ConnectAndLogin("127.0.0.1", 8000);
                float elapsed = 0f;
                while (ctx.Manager.CurrentState != ClientNetworkManager.State.InWorld && elapsed < 10f)
                {
                    ctx.Manager.Update(0.05f);
                    await System.Threading.Tasks.Task.Delay(50);
                    elapsed += 0.05f;
                }
                if (ctx.Manager.CurrentState != ClientNetworkManager.State.InWorld)
                    return (false, "login failed");

                // —— 观察 60s：bot 群在 ±500m 世界 8m/s 游走，静止观察者视野 80m ——
                // 统计口径（AOI 整链证据，非时序脆弱断言）：
                //   ① seenIds ≥ 2：生成路径工作（bot 走近被创建视图）
                //   ② 观察到 count 跌破峰值：销毁路径工作（bot 走远→停流→缺席超时删除）
                // 密度预期：100 bot / 1km² → 80m 视野期望 ~2 个；60s 位移 480m 必反复扫过视野。
                var seenIds = new System.Collections.Generic.HashSet<long>();
                int peakCount = 0;
                int dropCount = 0;
                for (int i = 0; i < 300; i++) // 300 × 200ms = 60s
                {
                    ctx.Manager.Update(0.05f);
                    await System.Threading.Tasks.Task.Delay(200);
                    int count = ctx.Registry.Count;
                    if (count > peakCount)
                    {
                        peakCount = count;
                    }
                    else if (count < peakCount)
                    {
                        dropCount++; // 视野内实体减少（离开→超时销毁）
                        peakCount = count;
                    }
                    foreach (var id in ctx.Registry.PlayerIds)
                        seenIds.Add(id);
                }
                if (seenIds.Count < 2)
                    return (false, $"observer saw only {seenIds.Count} distinct bot(s) in 60s (AOI spawn not working?)");
                if (dropCount == 0)
                    return (false, $"registry count never dropped below peak {peakCount} in 60s (AOI despawn/expiry not working?)");

                return (true, $"ok: saw {seenIds.Count} distinct bots, peak {peakCount}, {dropCount} drops");
            }
            finally
            {
                ctx.Manager.Dispose();
                Object.Destroy(root);
            }
        }
    }
```
